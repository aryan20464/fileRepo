#!/usr/bin/env bash
set -euo pipefail

INTERNAL_GITLAB="https://gitlab.internal"
INTERNAL_TOKEN="glpat-xxxxxxxxxxxxxxxxxxxx"   # internal PAT, scope: api

ORG_USER="orguser1"                            # organisational GitLab username
ORG_PASS="orgpass_or_token"                    # organisational GitLab password/PAT

INPUT_FILE="mirrors.csv"

while IFS=, read -r internal_url org_url; do
  # skip blank lines / comments
  [[ -z "$internal_url" || "$internal_url" == \#* ]] && continue

  internal_url=$(echo "$internal_url" | xargs)
  org_url=$(echo "$org_url" | xargs)

  relative_path=$(echo "$internal_url" | sed "s|${INTERNAL_GITLAB}/||" | sed 's|\.git$||')
  encoded_path=$(echo -n "$relative_path" | jq -sRr @uri)

  echo "==> Configuring mirror for $relative_path -> $org_url"

  RESP=$(curl -s --request POST --header "PRIVATE-TOKEN: $INTERNAL_TOKEN" \
       --form "url=$org_url" \
       --form "enabled=true" \
       --form "auth_method=password" \
       --form "username=$ORG_USER" \
       --form "password=$ORG_PASS" \
       "${INTERNAL_GITLAB}/api/v4/projects/${encoded_path}/remote_mirrors")

  mirror_id=$(echo "$RESP" | jq -r '.id // empty')
  if [[ -n "$mirror_id" ]]; then
    echo "   OK — mirror id: $mirror_id"
  else
    echo "   RESPONSE: $RESP"
  fi

  echo "-----------------------------------"
done < "$INPUT_FILE"
