# Create log directory and log files
RUN mkdir -p /var/lib/nginx/logs /tmp && \
    touch /var/lib/nginx/logs/access.log && \
    touch /var/lib/nginx/logs/error.log && \
    chown -R 65532:65532 /var/lib/nginx && \
    chmod 755 /var/lib/nginx/logs && \
    chmod 644 /var/lib/nginx/logs/access.log && \
    chmod 644 /var/lib/nginx/logs/error.log && \
    chown -R 65532:65532 /usr/share/nginx/html
