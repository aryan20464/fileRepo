apiVersion: v1
kind: ServiceAccount
metadata:
  name: app-scaler-sa
  namespace: engine
---
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  name: app-scaler-role
  namespace: engine
rules:
  - apiGroups: ["apps"]
    resources: ["deployments"]
    resourceNames: ["rate-engine-fixccilfeed", "rate-engine-fix360t"]
    verbs: ["get"]
  - apiGroups: ["apps"]
    resources: ["deployments/scale"]
    resourceNames: ["rate-engine-fixccilfeed", "rate-engine-fix360t"]
    verbs: ["get", "patch", "update"]
  # Needed for log capture and waiting for pods to start/stop.
  # Pod names are dynamic, so these cannot be limited with resourceNames.
  - apiGroups: [""]
    resources: ["pods"]
    verbs: ["get", "list", "watch"]
  - apiGroups: [""]
    resources: ["pods/log"]
    verbs: ["get"]
  - apiGroups: [""]
    resources: ["events"]
    verbs: ["get", "list"]
---
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: app-scaler-rb
  namespace: engine
subjects:
  - kind: ServiceAccount
    name: app-scaler-sa
    namespace: engine
roleRef:
  kind: Role
  name: app-scaler-role
  apiGroup: rbac.authorization.k8s.io
