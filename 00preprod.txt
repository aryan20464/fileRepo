import { ChangeDetectionStrategy, Component, DestroyRef, EventEmitter, Output, computed, effect, inject, signal } from '@angular/core';
import { DatePipe } from '@angular/common';
import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { interval, startWith, switchMap } from 'rxjs';
import { AuthService } from '../../core/auth/auth.service';
import { DashboardApiService } from './dashboard-api.service';
import { ConsumerFeedStatus } from './dashboard.models';

@Component({
  selector: 'app-dashboard',
  imports: [DatePipe],
  templateUrl: './dashboard.component.html',
  styleUrl: './dashboard.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class DashboardComponent {
  private readonly api = inject(DashboardApiService);
  private readonly destroyRef = inject(DestroyRef);
  readonly auth = inject(AuthService);
  readonly feeds = signal<ConsumerFeedStatus[]>([]);
  readonly serverTime = signal<string | null>(null);
  readonly loading = signal(false);
  readonly error = signal('');
  readonly notice = signal('');
  readonly stoppingId = signal<number | null>(null);
  readonly pendingEnableFeed = signal<ConsumerFeedStatus | null>(null);
  readonly liveCount = computed(() => this.feeds().filter(feed => feed.status === 'LIVE').length);
  readonly delayedCount = computed(() => this.feeds().filter(feed => feed.status === 'INTERMITTENT').length);
  readonly downCount = computed(() => this.feeds().filter(feed => feed.status === 'OFFLINE' || feed.status === 'UNAVAILABLE').length);
  @Output() readonly openPrices = new EventEmitter<ConsumerFeedStatus>();
  private started = false;

  constructor() {
    effect(() => {
      if (!this.auth.token() || this.started) return;
      this.started = true;
      interval(10000).pipe(
        startWith(0),
        switchMap(() => { this.loading.set(true); return this.api.feeds(); }),
        takeUntilDestroyed(this.destroyRef)
      ).subscribe({
        next: response => { this.feeds.set(response.consumers); this.serverTime.set(response.serverTime); this.error.set(''); this.loading.set(false); },
        error: error => { this.error.set(error.error?.message ?? error.message ?? 'Unable to load feed status.'); this.loading.set(false); this.started = false; }
      });
    });
  }

  refresh(): void {
    if (!this.auth.token() || this.loading()) return;
    this.loading.set(true);
    this.api.feeds().pipe(takeUntilDestroyed(this.destroyRef)).subscribe({
      next: response => { this.feeds.set(response.consumers); this.serverTime.set(response.serverTime); this.error.set(''); this.loading.set(false); },
      error: error => { this.error.set(error.error?.message ?? error.message ?? 'Unable to load feed status.'); this.loading.set(false); }
    });
  }

  statusLabel(feed: ConsumerFeedStatus): string {
    return ({ LIVE: 'Live', INTERMITTENT: 'Intermittent', OFFLINE: 'Not live', UNAVAILABLE: 'Unavailable' })[feed.status];
  }

  ageLabel(seconds: number | null): string {
    if (seconds === null) return 'No timestamp';
    if (seconds < 1) return 'Current';
    if (seconds < 60) return `${seconds}s ago`;
    if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`;
    if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`;
    return `${Math.floor(seconds / 86400)}d ago`;
  }

  toggleStreaming(feed: ConsumerFeedStatus): void {
    if (feed.status === 'UNAVAILABLE' || this.stoppingId() !== null) return;
    if (feed.allowStreamYn !== 'Y') {
      this.pendingEnableFeed.set(feed);
      return;
    }
    this.performStreamingToggle(feed);
  }

  confirmEnableStreaming(): void {
    const pending = this.pendingEnableFeed();
    this.pendingEnableFeed.set(null);
    const current = pending ? this.feeds().find(feed => feed.consumerId === pending.consumerId) : null;
    if (current && current.allowStreamYn !== 'Y' && this.stoppingId() === null) this.performStreamingToggle(current);
  }

  cancelEnableStreaming(): void {
    this.pendingEnableFeed.set(null);
  }

  private performStreamingToggle(feed: ConsumerFeedStatus): void {
    this.error.set('');
    this.notice.set('');
    this.stoppingId.set(feed.consumerId);
    this.api.toggleStreaming(feed.consumerId).pipe(takeUntilDestroyed(this.destroyRef)).subscribe({
      next: response => {
        this.feeds.update(rows => rows.map(row => row.consumerId === feed.consumerId ? { ...row, allowStreamYn: response.allowStreamYn } : row));
        this.notice.set(`${response.consumerName} streaming has been ${response.allowStreamYn === 'Y' ? 'enabled' : 'disabled'}.`);
        this.stoppingId.set(null);
      },
      error: error => {
        this.error.set(error.error?.message ?? error.message ?? 'Streaming toggle failed.');
        this.stoppingId.set(null);
      }
    });
  }

  viewPrices(feed: ConsumerFeedStatus): void {
    if (feed.status !== 'UNAVAILABLE' && this.stoppingId() === null) this.openPrices.emit(feed);
  }
}


  <section class="dashboard-page">
  <header class="dashboard-heading">
    <div>
      <span>MARKET DATA OBSERVABILITY</span>
      <h1>Consumer Feed Dashboard</h1>
      <p>Real-time feed status for approved consumers.</p>
    </div>
    <div class="dashboard-actions">
      @if (serverTime()) { <small>Server time<br><strong>{{ serverTime() | date:'dd MMM yyyy · HH:mm:ss' }}</strong></small> }
      <button type="button" (click)="refresh()" [disabled]="loading()"><span>↻</span>{{ loading() ? 'Refreshing' : 'Refresh' }}</button>
    </div>
  </header>

  <div class="health-strip">
    <article class="health-stat total"><span class="stat-icon grid-icon"></span><div><strong>{{ feeds().length }}</strong><small>Approved consumers</small></div></article>
    <article class="health-stat live"><span class="stat-icon signal-icon"></span><div><strong>{{ liveCount() }}</strong><small>Feeds live</small></div></article>
    <article class="health-stat delayed"><span class="stat-icon clock-icon"></span><div><strong>{{ delayedCount() }}</strong><small>Intermittent</small></div></article>
    <article class="health-stat down"><span class="stat-icon alert-icon">!</span><div><strong>{{ downCount() }}</strong><small>Unavailable / down</small></div></article>
  </div>

  @if (error()) { <div class="dashboard-error"><strong>Feed status unavailable</strong><span>{{ error() }}</span></div> }
  @if (notice()) { <div class="dashboard-notice"><strong>Streaming configuration updated</strong><span>{{ notice() }}</span></div> }

  @if (!auth.token()) {
    <div class="dashboard-empty"><span class="radar"></span><h2>Connect to view feed health</h2><p>Add a valid JWT token from the user menu.</p></div>
  } @else if (loading() && feeds().length === 0) {
    <div class="dashboard-empty"><span class="radar active"></span><h2>Scanning consumer feeds</h2><p>Reading the latest market-data heartbeat from Oracle.</p></div>
  } @else if (feeds().length === 0) {
    <div class="dashboard-empty"><span class="radar"></span><h2>No approved consumers</h2><p>Approved consumers will appear here automatically.</p></div>
  } @else {
    <div class="feed-grid">
      @for (feed of feeds(); track feed.consumerId) {
        <article class="feed-card" [class]="'feed-card ' + feed.status.toLowerCase()" [class.panic-enabled]="feed.status !== 'UNAVAILABLE'" [class.stopping]="stoppingId() === feed.consumerId" [attr.role]="feed.status !== 'UNAVAILABLE' ? 'button' : null" [attr.tabindex]="feed.status !== 'UNAVAILABLE' ? 0 : null" (click)="toggleStreaming(feed)" (keydown.enter)="toggleStreaming(feed)" (keydown.space)="feed.status !== 'UNAVAILABLE' && $event.preventDefault(); toggleStreaming(feed)">
          <div class="feed-card-top">
            <div class="feed-symbol" [class.antenna-symbol]="feed.status !== 'UNAVAILABLE'"><span></span><i></i><b></b></div>
            <span class="status-pill"><i></i>{{ statusLabel(feed) }}</span>
          </div>
          <div class="feed-identity"><h2>{{ feed.consumerName }}</h2><p>{{ feed.detail }}</p></div>
          <div class="feed-telemetry">
            <div><small>LAST VALIDITY</small><strong>{{ feed.validity ? (feed.validity | date:'dd MMM · HH:mm:ss.SSS') : 'Not available' }}</strong></div>
            <div><small>LATENCY</small><strong>{{ ageLabel(feed.ageSeconds) }}</strong></div>
          </div>
          <footer><span><i></i>{{ feed.tableAvailable ? 'Feed source connected' : 'Feed source unavailable' }}</span>@if (feed.status !== 'UNAVAILABLE') { <div class="card-controls"><button type="button" class="view-prices" (click)="$event.stopPropagation(); viewPrices(feed)">View prices</button><small [class.streaming-enabled]="feed.allowStreamYn === 'Y'" [class.streaming-disabled]="feed.allowStreamYn !== 'Y'">{{ stoppingId() === feed.consumerId ? 'Updating…' : ('Streaming ' + (feed.allowStreamYn === 'Y' ? 'enabled' : 'disabled')) }}</small></div> }</footer>
        </article>
      }
    </div>
  }

  @if (pendingEnableFeed(); as feed) {
    <div class="confirmation-backdrop" role="presentation" (click)="cancelEnableStreaming()">
      <section class="confirmation-dialog" role="dialog" aria-modal="true" aria-labelledby="enable-streaming-title" aria-describedby="enable-streaming-message" (click)="$event.stopPropagation()">
        <span class="confirmation-icon" aria-hidden="true">!</span>
        <h2 id="enable-streaming-title">Enable streaming?</h2>
        <p id="enable-streaming-message">Feed delivery for <strong>{{ feed.consumerName }}</strong> will resume immediately.</p>
        <div class="confirmation-actions">
          <button type="button" class="cancel-action" (click)="cancelEnableStreaming()">Cancel</button>
          <button type="button" class="enable-action" (click)="confirmEnableStreaming()">Enable streaming</button>
        </div>
      </section>
    </div>
  }
</section>




import { HttpClient } from '@angular/common/http';
import { Injectable } from '@angular/core';
import { Observable } from 'rxjs';
import { DashboardResponse, StreamingToggleResponse } from './dashboard.models';

@Injectable({ providedIn: 'root' })
export class DashboardApiService {
  constructor(private readonly http: HttpClient) {}
  feeds(): Observable<DashboardResponse> { return this.http.get<DashboardResponse>('/api/dashboard/feeds'); }
  toggleStreaming(consumerId: number): Observable<StreamingToggleResponse> {
    return this.http.post<StreamingToggleResponse>(`/api/dashboard/feeds/${consumerId}/toggle`, {});
  }
}
export type FeedState = 'LIVE' | 'INTERMITTENT' | 'OFFLINE' | 'UNAVAILABLE';

export interface ConsumerFeedStatus {
  consumerId: number;
  consumerName: string;
  allowStreamYn: string;
  tableName: string | null;
  tableAvailable: boolean;
  status: FeedState;
  validity: string | null;
  ageSeconds: number | null;
  detail: string;
}

export interface DashboardResponse {
  serverTime: string;
  consumers: ConsumerFeedStatus[];
}

export interface StreamingToggleResponse {
  consumerId: number;
  consumerName: string;
  allowStreamYn: 'Y' | 'N';
  changedBy: string;
  changedAt: string;
}
