import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

@Repository
public class FIXConfigRepository {

    @Autowired
    private JdbcTemplate jdbcTemplate;

    public List<StaticDataDTO> findStaticDataEntry() {
        String sql = "SELECT LABEL, VALUE, MODULE FROM RE_FIX_STATICDATA WHERE MODULE = 'RE_RATEENGINE_API'";

        return jdbcTemplate.query(sql, (rs, rowNum) -> new StaticDataDTO(
                rs.getString("LABEL"),
                rs.getString("VALUE"),
                rs.getString("MODULE")
        ));
    }
}
