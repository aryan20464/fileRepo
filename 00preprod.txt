FROM cgr.dev/chainguard/nginx:latest

ENV TZ=Asia/Calcutta

# Switch to root to configure timezone, custom files, and permissions
USER root

# Install tzdata first since Chainguard images don't include it by default
# Then set up the timezone files correctly
RUN apk add --no-cache tzdata && \
    cp /usr/share/zoneinfo/$TZ /etc/localtime && \
    echo $TZ > /etc/timezone && \
    apk del tzdata

# Remove default site configs and copy your custom NGINX configuration
RUN rm -f /etc/nginx/conf.d/default.conf
COPY nginx preprod.conf /etc/nginx/nginx.conf

# Clean default HTML files and copy your built Angular application
RUN rm -rf /usr/share/nginx/html/*
COPY index.html /usr/share/nginx/html/
COPY 50x.html /usr/share/nginx/html/
COPY dist/ /usr/share/nginx/html/

# Ensure all NGINX paths are writable by the unprivileged user (UID 65532)
RUN chown -R 65532:65532 /usr/share/nginx/html \
    && chown -R 65532:65532 /etc/nginx \
    && chown -R 65532:65532 /var/cache/nginx \
    && chown -R 65532:65532 /var/log/nginx \
    && chown -R 65532:65532 /var/run

# Drop privileges back to the default non-root user
USER 65532

EXPOSE 4200

CMD ["nginx", "-g", "daemon off;"]
