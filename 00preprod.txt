import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import java.util.List;

@Configuration
public class SecretKeyConfig {

    @Bean(name = "jwtSecretKey")
    public String jwtSecretKey(FIXConfigRepository fixConfigRepository) {
        List<StaticDataDTO> entries = fixConfigRepository.findStaticDataEntry();
        
        return entries.stream()
                .filter(dto -> "JWT_TOKEN".equalsIgnoreCase(dto.getLABEL()))
                .map(StaticDataDTO::getVALUE)
                .findFirst()
                .orElseThrow(() -> new IllegalStateException("JWT_TOKEN missing from database"));
    }
}


@Service
public class JwtService {

    @Autowired
    @Qualifier("jwtSecretKey")
    private String secretKey;

    public void generateToken() {
        System.out.println("Using secret key directly: " + secretKey);
    }
}
