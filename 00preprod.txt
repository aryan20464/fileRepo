package com.sbi.rateengine.security;

public record CurrentUser(String userId, String userName, String entity, String role) {
    public boolean isChecker() {
        return "CHECKER".equalsIgnoreCase(role);
    }

    public boolean isMaker() {
        return "MAKER".equalsIgnoreCase(role);
    }
}


package com.sbi.rateengine.security;

import com.sbi.rateengine.consumer.InvalidRequestException;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.core.Authentication;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.stereotype.Service;
import org.springframework.security.access.AccessDeniedException;

@Service
public class CurrentUserService {
    private static final java.util.regex.Pattern USER_ID = java.util.regex.Pattern.compile("[A-Za-z0-9_.-]{1,20}");

    public CurrentUser requiredUser() {
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        Object principal = authentication == null ? null : authentication.getPrincipal();
        if (!(principal instanceof Jwt jwt)) {
            throw new InvalidRequestException("A valid JWT is required");
        }
        String userId = requiredClaim(jwt, "user_id", 20);
        String userName = requiredClaim(jwt, "user_name", 100);
        String entity = requiredClaim(jwt, "entity", 100);
        String role = requiredClaim(jwt, "role", 20).toUpperCase(java.util.Locale.ROOT);
        if (!USER_ID.matcher(userId).matches()) {
            throw new InvalidRequestException("JWT claim 'user_id' contains unsupported characters");
        }
        if (!"MAKER".equals(role) && !"CHECKER".equals(role)) {
            throw new AccessDeniedException("JWT role is not authorized");
        }
        return new CurrentUser(userId, userName, entity, role);
    }

    public CurrentUser requiredChecker() {
        CurrentUser user = requiredUser();
        if (!user.isChecker()) {
            throw new AccessDeniedException("A user with the CHECKER role is required");
        }
        return user;
    }

    public CurrentUser requiredMaker() {
        CurrentUser user = requiredUser();
        if (!user.isMaker()) {
            throw new AccessDeniedException("A user with the MAKER role is required");
        }
        return user;
    }

    private String requiredClaim(Jwt jwt, String name, int maxLength) {
        String value = jwt.getClaimAsString(name);
        if (value == null || value.isBlank()) {
            throw new InvalidRequestException("JWT claim '" + name + "' is required");
        }
        value = value.trim();
        if (value.length() > maxLength || value.chars().anyMatch(Character::isISOControl)) {
            throw new InvalidRequestException("JWT claim '" + name + "' is invalid");
        }
        return value;
    }
}

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.time.Instant;
import java.nio.charset.StandardCharsets;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicInteger;
import org.springframework.http.MediaType;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

@Component
public class MutationRateLimitFilter extends OncePerRequestFilter {
    private static final int MAX_MUTATIONS_PER_MINUTE = 120;
    private static final long WINDOW_SECONDS = 60;
    private static final long MAX_CONTENT_LENGTH = 1_048_576;
    private final ConcurrentHashMap<String, Window> windows = new ConcurrentHashMap<>();

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain chain)
            throws ServletException, IOException {
        if (request.getContentLengthLong() > MAX_CONTENT_LENGTH) {
            reject(response, 413, "{\"message\":\"Request payload is too large\"}");
            return;
        }
        if (isMutation(request) && !allow(request)) {
            response.setHeader("Retry-After", "60");
            reject(response, 429, "{\"message\":\"Too many mutation requests; retry later\"}");
            return;
        }
        chain.doFilter(request, response);
    }

    private boolean allow(HttpServletRequest request) {
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        String principal = authentication == null ? "anonymous" : authentication.getName();
        String key = principal + '|' + request.getRemoteAddr();
        long now = Instant.now().getEpochSecond();
        Window window = windows.compute(key, (ignored, current) -> {
            if (current == null || now - current.startedAt >= WINDOW_SECONDS) return new Window(now);
            current.count.incrementAndGet();
            return current;
        });
        if (windows.size() > 10_000) windows.entrySet().removeIf(entry -> now - entry.getValue().startedAt >= WINDOW_SECONDS);
        return window.count.get() <= MAX_MUTATIONS_PER_MINUTE;
    }

    private boolean isMutation(HttpServletRequest request) {
        String method = request.getMethod();
        return request.getRequestURI().startsWith("/api/")
                && ("POST".equals(method) || "PUT".equals(method) || "PATCH".equals(method) || "DELETE".equals(method));
    }

    private void reject(HttpServletResponse response, int status, String jsonBody) throws IOException {
        response.setStatus(status);
        response.setContentType(MediaType.APPLICATION_JSON_VALUE);
        response.setCharacterEncoding(StandardCharsets.UTF_8.name());
        response.getOutputStream().write(jsonBody.getBytes(StandardCharsets.UTF_8));
    }

    private static final class Window {
        private final long startedAt;
        private final AtomicInteger count = new AtomicInteger(1);
        private Window(long startedAt) { this.startedAt = startedAt; }
    }
}
