set -uo pipefail  # NOTE: -e intentionally omitted; errors handled explicitly below

REAL_BIN="/data/nginx/sbin/nginx.real"
CONF="/nginx/server/conf/nginx.conf"
LOCK="/data/nginx/.nginx_wrapper.lock"
LOG="/data/nginx/wrapper.log"

KA_DISPLAY="15";   RATE_DISPLAY="5r/s"
KA_RUNTIME="120";  RATE_RUNTIME="500r/s"

log() { echo "$(date '+%F %T') [nginx-wrapper] $*" >> "$LOG"; }

acquire_lock() {
    exec 9>"$LOCK"
    flock -n 9 || { echo "Another nginx wrapper invocation is in progress." >&2; exit 1; }
}
release_lock() {
    exec 9>&-
}

acquire_lock

# Track whether the file currently holds runtime values, so the EXIT trap
# knows whether it needs to revert.
SWAPPED=0

swap_to() {
    local ka="$1" rate="$2"
    # '#' delimiter avoids collision with the '/' inside values like "500r/s"
    sed -i -E "s#^(\s*keepalive_timeout\s+)[^;]+;#\1${ka};#" "$CONF"
    local rc1=$?
    sed -i -E "s#(rate=)[0-9]+r/s;#\1${rate};#" "$CONF"
    local rc2=$?
    if [[ $rc1 -ne 0 || $rc2 -ne 0 ]]; then
        log "ERROR: swap_to(${ka}, ${rate}) failed (rc1=$rc1 rc2=$rc2)"
        return 1
    fi
    return 0
}

# Safety net: if the script exits for ANY reason while the file is holding
# runtime values, force it back to display values before the shell exits.
cleanup_trap() {
    local exit_code=$?
    if [[ "$SWAPPED" -eq 1 ]]; then
        log "EXIT TRAP: script exiting (code $exit_code) with config still swapped -- forcing revert"
        swap_to "$KA_DISPLAY" "$RATE_DISPLAY" || log "EXIT TRAP: revert itself FAILED -- manual intervention required on $CONF"
    fi
}
trap cleanup_trap EXIT

IS_STATE_CHANGE=0
case "$*" in
    ""|"-s reload")
        IS_STATE_CHANGE=1
        ;;
    *)
        IS_STATE_CHANGE=0
        ;;
esac

if [[ "$IS_STATE_CHANGE" -eq 1 ]]; then
    log "State-changing op ('$*'): swapping to runtime values (keepalive_timeout=${KA_RUNTIME}, rate=${RATE_RUNTIME})"
    if ! swap_to "$KA_RUNTIME" "$RATE_RUNTIME"; then
        log "ERROR: initial swap to runtime values failed, aborting (trap will revert)"
        exit 1
    fi
    SWAPPED=1

    if ! "$REAL_BIN" -t -c "$CONF" >> "$LOG" 2>&1; then
        log "ERROR: config test failed with runtime values, aborting (trap will revert)"
        exit 1
    fi

    log "Executing real nginx with args: $*"
    # Release the lock BEFORE invoking real nginx. A bare start daemonizes
    # (double-forks and detaches), and if fd 9 is still open at that point
    # the detached master inherits it and holds the lock open forever --
    # every future invocation (even harmless -t) would then fail with
    # "Another nginx wrapper invocation is in progress." permanently.
    release_lock
    "$REAL_BIN" "$@"
    RC=$?

    # Reacquire for the confirm/revert step. If another wrapper invocation
    # is genuinely mid-swap right now, this will fail fast rather than
    # racing it -- acceptable, since that should be rare in practice.
    acquire_lock

    sleep 1
    if pgrep -f "nginx: master process" > /dev/null; then
        log "Reload/start confirmed active, reverting config to display values (keepalive_timeout=${KA_DISPLAY}, rate=${RATE_DISPLAY})"
        swap_to "$KA_DISPLAY" "$RATE_DISPLAY"
        SWAPPED=0
    else
        log "WARNING: master process not detected after operation; NOT reverting, leaving runtime values in file for inspection"
    fi

    exit $RC
else
    log "Read-only op ('$*'): passing through, no config mutation"
    SWAPPED=0
    release_lock
    "$REAL_BIN" "$@"
    exit $?
fi
