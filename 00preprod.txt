# Use only NGINX, no Node or NPM involved
FROM cgr.dev/chainguard/nginx:latest-nodev
ENV TZ=Asia/Calcutta
#RUN cp /usr/share/zoneinfo/$TZ /etc/localtime
RUN ln -snf /usr/share/zoneinfo/$TZ /etc/localtime && echo $TZ > /etc/timezone
# Start as root to fix permissions
USER root

# Remove default site config and add your own (optional)
RUN rm /etc/nginx/conf.d/default.conf
COPY nginx-preprod.conf /etc/nginx/nginx.conf

# Remove default site html and add your own to know the nginx is working in case u r using multiple projects (optional)
RUN rm /usr/share/nginx/html/index.html
COPY index.html /usr/share/nginx/html/
COPY 50x.html /usr/share/nginx/html/

# Copy the already-built Angular app into the image
COPY dist/ /usr/share/nginx/html/

# Set proper ownership to nginx user (UID 65532)
RUN chown -R 65532:65532 /usr/share/nginx/html && \
    chown -R 65532:65532 /etc/nginx && \
    chown -R 65532:65532 /var/cache/nginx && \
    chown -R 65532:65532 /var/log/nginx

# Drop to non-root user
USER 65532

EXPOSE 4200

CMD ["nginx", "-g", "daemon off;"]
