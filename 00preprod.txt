/* ============================================================
   CSV PARSER — handles quotes, embedded commas, CRLF and Excel's BOM
   ============================================================ */
function parseCSV(text){
  text = text.replace(/^\uFEFF/, "");
  var rows = [], row = [], field = "", inQ = false, i = 0, c;
  for (; i < text.length; i++){
    c = text[i];
    if (inQ){
      if (c === '"'){
        if (text[i+1] === '"'){ field += '"'; i++; }
        else inQ = false;
      } else field += c;
    } else if (c === '"'){ inQ = true; }
    else if (c === ","){ row.push(field); field = ""; }
    else if (c === "\n"){ row.push(field); rows.push(row); row = []; field = ""; }
    else if (c === "\r"){ /* skip */ }
    else field += c;
  }
  if (field.length || row.length){ row.push(field); rows.push(row); }
  return rows.filter(function(r){ return r.some(function(v){ return v.trim() !== ""; }); });
}

/* tolerant header mapping — column order in the CSV does not matter */
var HEADER_ALIASES = {
  app:"app", application:"app", module:"app", project:"app",
  host:"host", hostname:"host", vm:"host", vmname:"host", server:"host", servername:"host",
  role:"role", purpose:"role", function:"role",
  address:"ip", ip:"ip", ipaddress:"ip", ipaddr:"ip",
  environment:"env", env:"env",
  vlan:"vlan", vlanid:"vlan",
  servertype:"type", type:"type",
  platform:"os", os:"os", operatingsystem:"os",
  site:"site", datacentre:"site", datacenter:"site", dc:"site", location:"site",
  state:"state", status:"state"
};
function normHeader(h){ return String(h).toLowerCase().replace(/[^a-z]/g, ""); }

function normState(v){
  var s = String(v || "").toLowerCase();
  if (/down|stop|off|halt|fail/.test(s)) return "down";
  if (/maint|patch|hold|planned/.test(s)) return "maint";
  return "up";
}

function toRecords(rows){
  var head = rows[0].map(normHeader).map(function(h){ return HEADER_ALIASES[h] || h; });
  return rows.slice(1).map(function(r){
    var o = {};
    head.forEach(function(k, i){ o[k] = (r[i] || "").trim(); });
    o.app   = o.app   || "Unassigned";
    o.host  = o.host  || "(unnamed)";
    o.role  = o.role  || "";
    o.ip    = o.ip    || "";
    o.env   = (o.env  || "OTHER").toUpperCase();
    o.vlan  = o.vlan  || "";
    o.type  = o.type  || "";
    o.os    = o.os    || "";
    o.site  = o.site  || "";
    o.state = normState(o.state);
    return o;
  });
}

      render(true);
      countUp();
    })
    .catch(function(err){ fail(err); })
    .then(function(){ btn.classList.remove("spin"); });
}

      : "Check that <code>" + esc(CSV_FILE) + "</code> sits beside this page and that nginx can read it.") +
    "</span>";
}

function buildEnvironments(){
  var seen = [];
  SERVERS.forEach(function(s){ if (seen.indexOf(s.env) < 0) seen.push(s.env); });
  seen.sort(function(a,b){
    var ia = ENV_ORDER.indexOf(a), ib = ENV_ORDER.indexOf(b);
    if (ia < 0) ia = 99; if (ib < 0) ib = 99;
    return ia - ib || cmp(a,b);
  });
  ENVIRONMENTS = seen.map(function(code, i){
    var meta = ENV_META[code] || {};
    return {
      id: code,
      label: meta.label || code,
      note: meta.note || "",
      colour: "var(--env-" + ((i % 6) + 1) + ")",
      rank: i
    };
  });
  envById = {};
  ENVIRONMENTS.forEach(function(e){ envById[e.id] = e; });
}

function buildRail(){
  var rail = $("#rail");
  rail.innerHTML = "<div class='rail-head'><span>Environments</span><b id='railTotal'>" + SERVERS.length + "</b></div>";

  ENVIRONMENTS.forEach(function(env){
    var hosts = SERVERS.filter(function(s){ return s.env === env.id; });
    var b = document.createElement("button");
    b.className = "band"; b.type = "button";
    b.style.setProperty("--tick", env.colour);
    b.setAttribute("aria-pressed", "false");
    b.dataset.env = env.id;
    b.innerHTML =
      "<span class='band-top'><span class='band-name'>" + esc(env.label) + "</span>" +
      "<span class='band-count'>" + hosts.length + "</span></span>" +
      (env.note ? "<span class='band-note'>" + esc(env.note) + "</span>" : "") +
      "<span class='glyph'>" + hosts.map(function(h, i){
        return "<i data-state='" + h.state + "' style='--d:" + (120 + i * 26) + "ms'></i>";
      }).join("") + "</span>";
    b.addEventListener("click", function(){
      filterEnv = (filterEnv === env.id) ? null : env.id;
      syncRail(); render(true);
    });
    rail.appendChild(b);
  });

  var reset = document.createElement("button");
  reset.className = "rail-reset"; reset.type = "button";
  reset.textContent = "Show every environment";
  reset.addEventListener("click", function(){ filterEnv = null; syncRail(); render(true); });
  rail.appendChild(reset);

  var legend = document.createElement("div");
  legend.className = "legend";
  legend.innerHTML = "<span><i class='sw'></i>Running</span>" +
                     "<span><i class='sw' style='opacity:.35'></i>Under maintenance</span>" +
                     "<span><i class='sw hollow'></i>Stopped</span>";
  rail.appendChild(legend);
}
function syncRail(){
  Array.prototype.forEach.call(document.querySelectorAll(".band"), function(b){
    b.setAttribute("aria-pressed", String(b.dataset.env === filterEnv));
  });
}

/* ============================================================
   SORT + RENDER
   ============================================================ */
function defaultCmp(a, b){
  return cmp(a.app.toLowerCase(), b.app.toLowerCase())
      || cmp(a.role.toLowerCase(), b.role.toLowerCase())
      || cmp(ipNum(a.ip), ipNum(b.ip));
}

function visible(){
  var q = query.trim().toLowerCase();
  var list = SERVERS.filter(function(s){
    if (filterEnv && s.env !== filterEnv) return false;
    if (!q) return true;
    return [s.app, s.host, s.role, s.ip, s.vlan, s.type, s.os, s.site,
            (envById[s.env] || {}).label || s.env].join(" ").toLowerCase().indexOf(q) > -1;
  });

  if (!sort) return list.sort(defaultCmp);

  var k = sort.key, d = sort.dir;
  return list.sort(function(a, b){
    var x, y;
    if (k === "ip")        { x = ipNum(a.ip); y = ipNum(b.ip); }
    else if (k === "vlan") { x = parseInt(a.vlan,10) || 0; y = parseInt(b.vlan,10) || 0; }
    else if (k === "env")  { x = (envById[a.env]||{}).rank; y = (envById[b.env]||{}).rank; }
    else if (k === "state"){ x = ["down","maint","up"].indexOf(a.state); y = ["down","maint","up"].indexOf(b.state); }
    else                   { x = String(a[k]).toLowerCase(); y = String(b[k]).toLowerCase(); }
    return cmp(x, y) * d || defaultCmp(a, b);
  });
}

function render(animate){
  var list = visible();
  rowsEl.classList.toggle("stagger", !!animate);

  var lastApp = null;
  rowsEl.innerHTML = list.map(function(s, i){
    var env = envById[s.env] || { label:s.env, colour:"var(--ink-faint)" };
    var isNew = s.app !== lastApp;
    var groupable = !sort;              // only draw the divider when the default app order is active
    lastApp = s.app;
    return "<tr data-env='" + esc(s.env) + "' style='--tick:" + env.colour + ";--d:" + (Math.min(i,16)*20) + "ms'" +
           (groupable && isNew && i > 0 ? " class='appbreak'" : "") + ">" +
      "<td><span class='app'>" + esc(s.app) + "</span></td>" +
      "<td><span class='host'>" + esc(s.host) + "</span>" +
          (s.role ? "<span class='role'>" + esc(s.role) + "</span>" : "") + "</td>" +
      "<td><button class='ip' data-ip='" + esc(s.ip) + "' title='Copy address'>" + esc(s.ip) + "</button></td>" +
      "<td><span class='envtag' style='color:" + env.colour + "'>" + esc(env.label) + "</span></td>" +
      "<td><span class='vlan'>" + esc(s.vlan) + "</span></td>" +
      "<td><span class='tcell'>" + typeIcon(s.type) + esc(s.type) + "</span></td>" +
      "<td><span class='tcell os'>" + osIcon(s.os) + esc(s.os) + "</span></td>" +
      "<td>" + esc(s.site) + "</td>" +
      "<td><span class='state' data-state='" + s.state + "'>" + STATE_LABEL[s.state] + "</span></td>" +
    "</tr>";
  }).join("");

  if (!list.length){
    noticeEl.hidden = false;
    noticeEl.innerHTML = "<b>No hosts match that filter</b>Clear the search box or pick a different environment.";
  } else {
    noticeEl.hidden = true;
  }
  $("#shown").textContent = list.length;
  $("#total").textContent = SERVERS.length;
}

/* ---- header sorting; a third click returns to the default order ---- */
Array.prototype.forEach.call(document.querySelectorAll("thead th"), function(th){
  th.addEventListener("click", function(){
    var key = th.dataset.key;
    if (sort && sort.key === key && sort.dir === -1) sort = null;
    else if (sort && sort.key === key) sort = { key:key, dir:-1 };
    else sort = { key:key, dir:1 };

    Array.prototype.forEach.call(document.querySelectorAll("thead th"), function(o){
      o.removeAttribute("aria-sort");
      o.querySelector(".arrow").textContent = "";
    });
    if (sort){
      th.setAttribute("aria-sort", sort.dir === 1 ? "ascending" : "descending");
      th.querySelector(".arrow").textContent = sort.dir === 1 ? "\u25B2" : "\u25BC";
    }
    render();
  });
});

/* ---- copy address (works over plain HTTP, where the async clipboard API is blocked) ---- */
rowsEl.addEventListener("click", function(e){
  var btn = e.target.closest ? e.target.closest(".ip") : null;
  if (!btn) return;
  copy(btn.dataset.ip);
  btn.classList.add("copied");
  setTimeout(function(){ btn.classList.remove("copied"); }, 1100);
  toast(btn.dataset.ip + " copied");
});
function copy(text){
  function fallback(){
    var ta = document.createElement("textarea");
    ta.value = text; ta.setAttribute("readonly", "");
    ta.style.cssText = "position:fixed;top:-1000px;opacity:0";
    document.body.appendChild(ta); ta.select();
    try { document.execCommand("copy"); } catch (_){}
    ta.parentNode.removeChild(ta);
  }
  if (navigator.clipboard && window.isSecureContext) navigator.clipboard.writeText(text).catch(fallback);
  else fallback();
}
var toastTimer;
function toast(msg){
  var t = $("#toast");
  t.textContent = msg; t.classList.add("show");
  clearTimeout(toastTimer);
  toastTimer = setTimeout(function(){ t.classList.remove("show"); }, 1400);
}

/* ---- counts tick up once ---- */
function countUp(){
  if (window.matchMedia && matchMedia("(prefers-reduced-motion:reduce)").matches) return;
  Array.prototype.forEach.call(document.querySelectorAll(".band-count, #railTotal"), function(el){
    var target = parseInt(el.textContent, 10);
    if (!target) return;
    var start = performance.now(), dur = 620;
    el.textContent = "0";
    (function step(now){
      var p = Math.min((now - start) / dur, 1);
      el.textContent = Math.round(target * (1 - Math.pow(1 - p, 3)));
      if (p < 1) requestAnimationFrame(step);
    })(start);
  });
}
