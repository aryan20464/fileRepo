        runAsNonRoot: true
        runAsUser: 101
        fsGroup: 101                 # lets UID 101 read the key from the Secret volume
        seccompProfile:
          type: RuntimeDefault
      containers:
        - name: rateengine-ui
          image: h05vksharbor.corp.ad.com/itss/rateengine-ui:2.2-monitor-ns

            allowPrivilegeEscalation: false

              drop:
                - ALL
          volumeMounts:
            - name: nginx-conf
              mountPath: /etc/nginx/nginx.conf
              subPath: nginx.conf
              readOnly: true
            - name: tls
              mountPath: /etc/nginx/tls
              readOnly: true
      volumes:
        - name: nginx-conf
          configMap:
            name: rateengine-ui-nginx
        - name: tls

openssl pkcs12 -in rateengine.p12 -clcerts -nokeys -out leaf.crt
openssl pkcs12 -in rateengine.p12 -cacerts -nokeys -out chain.crt
openssl pkcs12 -in rateengine.p12 -nocerts -nodes -out tls.key
cat leaf.crt chain.crt > tls.crt      # server cert first, then intermediates
chmod 600 tls.key
