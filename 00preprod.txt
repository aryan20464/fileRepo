import { Inject, Injectable, DOCUMENT } from "@angular/core";
import { BehaviorSubject, Observable, of, throwError } from "rxjs";
import { catchError, mapTo, tap, timeout } from "rxjs/operators";
import { HttpClient, HttpHeaders } from "@angular/common/http";
import { Tokens } from "./tokens";
import { LoginUser } from "./LoginUser";
import { Role } from "./roles";
import { jwtDecode } from "jwt-decode";
import {
  environment,

  rateconst,
} from "../../environments/environment";

import { CommonService } from "./common.service";

@Injectable({
  providedIn: "root",
})
export class AuthService {
  baseUrl = rateconst.rforexAuthServiceApiUrl;
  initVector = rateconst.initVector;
  private readonly JWT_TOKEN = "JWT_TOKEN";
  private readonly REFRESH_TOKEN = "REFRESH_TOKEN";
  private readonly USER_DETAILS = "USER";
  private readonly sessionKey = "activeSession";
  isLoggedIn$ = new BehaviorSubject<boolean>(false);
  userPfId: any;
  uniqueId: string = "";
  isTimeOut: boolean = false;
  private currentUrl = "";
  token: string = "";
  headers = new HttpHeaders().set("Content-Type", "application/json");

  constructor(
    public http: HttpClient,
    public router: Router,
    public commonService: CommonService,
    @Inject(DOCUMENT) private document: Document,
  ) {
    window.addEventListener("storage", (event) => {
      if (event.key === this.sessionKey) {
        this.userPfId = "";
      }
    });
  }

  user: LoginUser | undefined; 
  userdetails = {};


  getCurrentSession() {
    const val = localStorage.getItem(this.sessionKey);
    return val ? JSON.parse(val) : null;
  }

  loginCBS(pfId: any) {
    const sessionData = { pfId, timestamp: Date.now() };
    localStorage.setItem(this.sessionKey, JSON.stringify(sessionData));
    this.isLoggedIn$.next(true);
  }


  setCurrentUrl(url: string): void {
    this.currentUrl = url;
  }

  getCurrentUrl() {
    return this.currentUrl;
  }

  getMockUser() {
    return this.http.get(`${this.baseUrl}/public/getUser`);
  }

  getAllMockedUsers(): Observable<any> {
    return this.http.get(`${this.baseUrl}/public/getAllUsers`);
  }

  getUser() {
    if (!this.user) {
      let token: string = this.getJwtToken() ?? '';;
      const usr: LoginUser = (this.decodeJwtToken(token) as any)?.user;
      return usr;
    }

    return this.user;
  }

  setUser(usr: LoginUser) {
    this.user = usr;
  }

  async login(dataid: string): Promise<Observable<boolean>> {
    return this.http
      .get<any>(`${this.baseUrl}/public/getUserFromFXS/${dataid}`)
      .pipe(
        tap((userData) => {
          console.log("User data "+JSON.stringify(userData))
          this.user = userData.userName;
          console.log("user "+this.user)
          this.doLoginUser(
            userData.userName,
            userData.token,
            userData.refreshToken,
          );
        }),
        mapTo(true),
        catchError((error) => {
          alert("Unable to fetch User details");
          return of(false);
        }),
      );
  }

  logout() {
    return this.http
      .post<any>(`${this.baseUrl}/logout`, {
        refreshToken: this.getRefreshToken(),
      })
      .pipe(
        tap(() => this.doLogoutUser()),
        mapTo(true),
        catchError((error) => {
          alert(error.error);
          return of(false);
        }),
      );
  }


  isLoggedIn() {
    return !!this.getJwtToken();
  }

  refreshToken() {
    return this.http
      .post<any>(`${this.baseUrl}/public/refreshToken`, {
        refreshToken: this.getRefreshToken(),
      })
      .pipe(
        tap((tokens: Tokens) => {
          this.storeJwtToken(tokens.jwt);
        }),
      );
  }

  getJwtToken() {
    return sessionStorage.getItem(this.JWT_TOKEN);
  }

  decodeJwtToken(token: string) {
    try {
      return jwtDecode(token);
    } catch (Error) {
      return null;
    }
  }

  public doLoginUser(username: string, tokens: string, refreshToken: string) {
    const mytoken: Tokens = new Tokens(tokens, refreshToken);
    console.log("token in dologinuser "+JSON.stringify(mytoken))
    console.log("jwt token in dologinuser "+mytoken.jwt)
    this.storeJwtToken(mytoken.jwt);
  }

  public doLogoutUser() {
    console.log("do logout user")
    this.removeTokens();
    this.router.navigate(["login"]);
  }

  private getRefreshToken() {
    return sessionStorage.getItem(this.REFRESH_TOKEN);
  }

  public storeJwtToken(jwt: string ) {
    console.log("jwt in storejwtoken "+jwt)
    console.log("jwt null "+ jwt !=null)
    console.log("jwt length "+ jwt.trim.length)
    if (jwt != null && jwt.trim().length > 0) {
      console.log("JWT TOken "+this.JWT_TOKEN)
      console.log("Refresh TOken "+this.REFRESH_TOKEN)
      sessionStorage.setItem(this.JWT_TOKEN, jwt);
      sessionStorage.setItem(this.REFRESH_TOKEN, "refresh");
    }
    else{
      console.log("jwt is null in storejwttoken")
    }
  }

  private removeTokens() {
    console.log("removetokens")
    sessionStorage.removeItem(this.JWT_TOKEN);
    sessionStorage.removeItem(this.REFRESH_TOKEN);
  }

  isAuthorized() {
    return !!this.getUser();
  }
  hasRole(role: string): boolean {
    if (this.isAuthorized()) {
      let matched = false;
      this.getUser().roles.forEach((userrole: string) => {
        if (userrole === role) {
          matched = true;
        }
      });
      return matched;
    } else {
      return false;
    }
  }

  hasRole2(roles: Role[]): boolean {
    for (let i = 0; i < roles.length; i++) {
      if (this.isAuthorized()) {
        for (let j = 0; j < this.getUser().roles.length; j++) {
          if (this.getUser().roles[j].roleName === roles[i].roleName) {
            return true;
          }
        }
      } else {
        return false;
      }
    }
    return false;
  }

  checkuser() {
    localStorage.clear();

    const token = this.getJwtToken() ?? '';
    const tokenusernameValue: any = this.decodeJwtToken(token);
    this.user = new LoginUser(tokenusernameValue.user[0]);
    let userRoles: any = {
      role: tokenusernameValue.user[0].role,
    };
    const roleAttr = tokenusernameValue.user[0].roleAttr;
    for (const property in roleAttr) {
      userRoles[`${property}`] = `${roleAttr[property]}`;
    }
    let key: string;
    this.userdetails = userRoles;
    for (let i in this.userdetails) {
    }
    localStorage.setItem("username", tokenusernameValue.user[0].name);
    localStorage.setItem("emailId", tokenusernameValue.user[0].emailId);
    return userRoles;
  }
  
}

  readonly view = signal<'dashboard' | 'prices' | 'consumers' | 'mappings' | 'subscriptions' | 'spreads' | 'swaps'>('dashboard');
  readonly selectedConsumer = signal<ConsumerFeedStatus | null>(null);
  tokenDraft = '';
  constructor(readonly auth: AuthService) {}

  showPrices(consumer: ConsumerFeedStatus): void {
    if (consumer.status === 'UNAVAILABLE') return;
    this.selectedConsumer.set(consumer);
    this.view.set('prices');
  }

  connect(): void {
    const token = this.tokenDraft.trim();
    if (!token) { this.authError.set('Enter a JWT access token.'); return; }
    if (token.length > 8192 || token.split('.').length !== 3) {
      this.authError.set('The JWT access token format or size is invalid.');
      return;
    }
    this.auth.setToken(token);
    if (!this.auth.user()) {
      this.auth.clear();
      this.authError.set('This token is invalid or does not contain the required claims.');
      return;
    }
    this.authError.set('');
    this.tokenDraft = '';
    this.showAuth.set(false);
  }
}
