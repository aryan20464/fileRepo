readonly token = signal<string | null>(this.getJwtToken());
readonly user = computed(() => {
  const t = this.token();
  if (!t) return null;
  const decoded: any = this.decodeJwtToken(t);
  return decoded?.user ?? null; // matches backend's claims.put("user", user)
});
readonly roles = computed(() => {
  const t = this.token();
  if (!t) return [];
  const decoded: any = this.decodeJwtToken(t);
  return (decoded?.auth ?? []).map((a: any) => a?.authority).filter(Boolean);
});
readonly isMaker = computed(() => this.roles().includes('MAKER'));
readonly isChecker = computed(() => this.roles().includes('CHECKER'));


public storeJwtToken(jwt: string) {
  if (jwt != null && jwt.trim().length > 0) {
    sessionStorage.setItem(this.JWT_TOKEN, jwt);
    sessionStorage.setItem(this.REFRESH_TOKEN, "refresh");
    this.token.set(jwt);           // ← added
  }
}

private removeTokens() {
  sessionStorage.removeItem(this.JWT_TOKEN);
  sessionStorage.removeItem(this.REFRESH_TOKEN);
  this.token.set(null);            // ← added
}
