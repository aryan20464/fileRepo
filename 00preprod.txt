apiVersion: v1
kind: ConfigMap
metadata:
  name: holiday-config
data:
  holidays.txt: |
    2026-01-01
    2026-04-30
    2026-05-01
    2026-09-02





apiVersion: batch/v1
kind: CronJob
metadata:
  name: my-holiday-aware-job
spec:
  schedule: "0 9 * * 1-5" 
  jobTemplate:
    spec:
      template:
        spec:
          volumes:
          - name: holiday-volume
            configMap:
              name: holiday-config
          containers:
          - name: worker
            image: python:3.9-slim
            volumeMounts:
            - name: holiday-volume
              mountPath: /etc/holidays
            command: ["/bin/sh", "-c"]
            args:
            - |
              CURRENT_DATE=$(date +%Y-%m-%d)
              if grep -q "$CURRENT_DATE" /etc/holidays/holidays.txt; then
                echo "Skipping execution: $CURRENT_DATE is a listed holiday."
                exit 0
              fi
              echo "Proceeding with job execution..."
              python /app/main.py
          restartPolicy: OnFailure
