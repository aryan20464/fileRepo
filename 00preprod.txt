function typeIcon(type){
  const t = type.toLowerCase();
  if (t.includes("weblogic"))   return ICONS.weblogic;
  if (t.includes("tomcat"))     return ICONS.tomcat;
  if (t.includes("nginx"))      return ICONS.nginx;
  if (t.includes("oracle") || t.includes("postgres") || t.includes("db")) return ICONS.oracle;
  if (t.includes("kube") || t.includes("k8s") || t.includes("ocp"))       return ICONS.kubernetes;
  if (t.includes("gitlab") || t.includes("jenkins"))                      return ICONS.gitlab;
  if (t.includes("artifact") || t.includes("nexus"))                      return ICONS.artifact;
  return ICONS.box;
}
function osIcon(os){
  const o = os.toLowerCase();
  return (o.includes("rhel") || o.includes("oel") || o.includes("linux") || o.includes("ubuntu"))
    ? ICONS.linux : ICONS.chip;
}

let filterEnv = null;
let query = "";
let sort = { key:"host", dir:1 };

const $ = s => document.querySelector(s);
const rowsEl = $("#rows"), emptyEl = $("#empty");

function esc(s){ return String(s).replace(/[&<>"]/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[c])); }

function buildRail(){
  const rail = $("#rail");
  ENVIRONMENTS.forEach(env => {
    const hosts = SERVERS.filter(s => s.env === env.id);
    const b = document.createElement("button");
    b.className = "band";
    b.type = "button";
    b.style.setProperty("--tick", env.colour);
    b.setAttribute("aria-pressed", "false");
    b.dataset.env = env.id;
    b.innerHTML =
      `<span class="band-top"><span class="band-name">${esc(env.label)}</span>` +
      `<span class="band-count">${hosts.length}</span></span>` +
      `<span class="band-note">${esc(env.note)}</span>` +
      `<span class="glyph">${hosts.map((h, i) => `<i data-state="${h.state}" style="--d:${120 + i * 26}ms"></i>`).join("")}</span>`;
    b.addEventListener("click", () => {
      filterEnv = (filterEnv === env.id) ? null : env.id;
      syncRail(); render(true);
    });
    rail.appendChild(b);
  });

  const reset = document.createElement("button");
  reset.className = "rail-reset";
  reset.type = "button";
  reset.textContent = "Show every environment";
  reset.addEventListener("click", () => { filterEnv = null; syncRail(); render(true); });
  rail.appendChild(reset);

  const legend = document.createElement("div");
  legend.className = "legend";
  legend.innerHTML =
    `<span><i class="sw"></i>Running</span>` +
    `<span><i class="sw" style="opacity:.35"></i>Under maintenance</span>` +
    `<span><i class="sw hollow"></i>Stopped</span>`;
  rail.appendChild(legend);

  $("#railTotal").textContent = SERVERS.length;
}

function syncRail(){
  document.querySelectorAll(".band").forEach(b =>
    b.setAttribute("aria-pressed", String(b.dataset.env === filterEnv)));
}

function visible(){
  const q = query.trim().toLowerCase();
  return SERVERS
    .filter(s => !filterEnv || s.env === filterEnv)
    .filter(s => !q || [s.host, s.role, s.ip, s.vlan, s.type, s.os, s.site, envById[s.env].label]
      .join(" ").toLowerCase().includes(q))
    .sort((a, b) => {
      const k = sort.key;
      let x = k === "env" ? envById[a.env].label : (a[k] ?? "");
      let y = k === "env" ? envById[b.env].label : (b[k] ?? "");
      if (k === "ip") { x = ipNum(a.ip); y = ipNum(b.ip); }
      if (k === "vlan") { x = +a.vlan; y = +b.vlan; }
      return (x > y ? 1 : x < y ? -1 : 0) * sort.dir;
    });
}
function ipNum(ip){ return ip.split(".").reduce((n, o) => n * 256 + (+o || 0), 0); }

function render(animate){
  const list = visible();
  rowsEl.classList.toggle("stagger", !!animate);
  rowsEl.innerHTML = list.map((s, i) => {
    const env = envById[s.env];
    const d = Math.min(i, 16) * 20;
    return `<tr data-env="${s.env}" style="--tick:${env.colour};--d:${d}ms">
      <td><span class="host">${esc(s.host)}</span><span class="role">${esc(s.role)}</span></td>
      <td><button class="ip" data-ip="${esc(s.ip)}" title="Copy address">${esc(s.ip)}</button></td>
      <td><span class="envtag" style="--tick:${env.colour};color:${env.colour}">${esc(env.label)}</span></td>
      <td><span class="vlan">${esc(s.vlan)}</span></td>
      <td><span class="tcell">${typeIcon(s.type)}${esc(s.type)}</span></td>
      <td><span class="tcell os">${osIcon(s.os)}${esc(s.os)}</span></td>
      <td>${esc(s.site)}</td>
      <td><span class="state" data-state="${s.state}">${STATE_LABEL[s.state]}</span></td>
    </tr>`;
  }).join("");

  emptyEl.hidden = list.length > 0;
  $("#shown").textContent = list.length;
  $("#total").textContent = SERVERS.length;
}

/* ---- sorting ---- */
document.querySelectorAll("thead th").forEach(th => {
  th.addEventListener("click", () => {
    const key = th.dataset.key;
    sort = { key, dir: sort.key === key ? -sort.dir : 1 };
    document.querySelectorAll("thead th").forEach(o => {
      o.removeAttribute("aria-sort");
      o.querySelector(".arrow").textContent = "";
    });
    th.setAttribute("aria-sort", sort.dir === 1 ? "ascending" : "descending");
    th.querySelector(".arrow").textContent = sort.dir === 1 ? "▲" : "▼";
    render();
  });
});

/* ---- search ---- */
$("#q").addEventListener("input", e => { query = e.target.value; render(); });
document.addEventListener("keydown", e => {
  if (e.key === "/" && document.activeElement !== $("#q")) { e.preventDefault(); $("#q").focus(); }
  if (e.key === "Escape" && document.activeElement === $("#q")) { $("#q").value = ""; query = ""; render(); }
});

/* ---- copy address (works over plain HTTP, where the async clipboard API is blocked) ---- */
rowsEl.addEventListener("click", e => {
  const btn = e.target.closest(".ip");
  if (!btn) return;
  copy(btn.dataset.ip);
  btn.classList.add("copied");
  setTimeout(() => btn.classList.remove("copied"), 1100);
  toast(btn.dataset.ip + " copied");
});
function copy(text){
  if (navigator.clipboard && window.isSecureContext) {
    navigator.clipboard.writeText(text).catch(fallback);
  } else { fallback(); }
  function fallback(){
    const ta = document.createElement("textarea");
    ta.value = text;
    ta.setAttribute("readonly", "");
    ta.style.cssText = "position:fixed;top:-1000px;opacity:0";
    document.body.appendChild(ta);
    ta.select();
    try { document.execCommand("copy"); } catch (_) {}
    ta.remove();
  }
}
let toastTimer;
function toast(msg){
  const t = $("#toast");
  t.textContent = msg;
  t.classList.add("show");
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => t.classList.remove("show"), 1400);
}

/* ---- theme ---- */
$("#theme").addEventListener("click", () => {
  const root = document.documentElement;
  const dark = matchMedia("(prefers-color-scheme:dark)").matches;
  const now = root.dataset.theme === "auto" ? (dark ? "light" : "dark")
            : root.dataset.theme === "dark" ? "light" : "dark";
  root.dataset.theme = now;
});

/* counts tick up once, as the glyphs fill in */
function countUp(){
  if (matchMedia("(prefers-reduced-motion:reduce)").matches) return;
  document.querySelectorAll(".band-count, #railTotal").forEach(el => {
    const target = +el.textContent;
    if (!target) return;
    const start = performance.now(), dur = 620;
    el.textContent = "0";
    (function step(now){
      const p = Math.min((now - start) / dur, 1);
      el.textContent = Math.round(target * (1 - Math.pow(1 - p, 3)));
      if (p < 1) requestAnimationFrame(step);
    })(start);
  });
}
