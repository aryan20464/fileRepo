
# ============================================================
# STAGE 1: Prepare writable directories using BusyBox
# ============================================================
FROM busybox:1.37 AS filesystem

RUN mkdir -p /app/logs \
    && chown -R 65532:65532 /app \
    && chmod 755 /app \
    && chmod 775 /app/logs


# ============================================================
# STAGE 2: Production image using Chainguard JRE
# ============================================================
FROM cgr.dev/chainguard/jre:latest

ENV TZ=Asia/Calcutta

WORKDIR /app

# ============================================================
# Copy prepared directory with ownership preserved
# ============================================================
COPY --from=filesystem --chown=65532:65532 /app /app

# ============================================================
# Copy application certificates
# ============================================================
COPY --chown=65532:65532 src/main/resources/preprod/fxs-idea-preprod.crt /app/fxs-idea-preprod.crt

COPY --chown=65532:65532 src/main/resources/preprod/fxs-idea-uat.crt /app/fxs-idea-uat.crt

COPY --chown=65532:65532 src/main/resources/preprod/rforex-vks-preprod.crt /app/rforex-vks-preprod.crt

COPY --chown=65532:65532 src/main/resources/preprod/NBC_CBS_SSO_PublicKey-UAT.cer /app/NBC_CBS_SSO_PublicKey.cer

# ============================================================
# Copy Spring Boot WAR
# ============================================================
COPY --chown=65532:65532 target/rforexinterfaceservice-0.0.1-SNAPSHOT.war /app/rforexinterfaceservice.war

# ============================================================
# Import certificates into Java truststore
# ============================================================
RUN ["keytool", "-importcert", "-trustcacerts", "-cacerts", "-storepass", "changeit", "-noprompt", "-alias", "fxs-certs-preprod", "-file", "/app/fxs-idea-preprod.crt"]

RUN ["keytool", "-importcert", "-trustcacerts", "-cacerts", "-storepass", "changeit", "-noprompt", "-alias", "fxs-certs-uat", "-file", "/app/fxs-idea-uat.crt"]

RUN ["keytool", "-importcert", "-trustcacerts", "-cacerts", "-storepass", "changeit", "-noprompt", "-alias", "rfx-certs", "-file", "/app/rforex-vks-preprod.crt"]

RUN ["keytool", "-importcert", "-trustcacerts", "-cacerts", "-storepass", "changeit", "-noprompt", "-alias", "cbs-sso-certs", "-file", "/app/NBC_CBS_SSO_PublicKey.cer"]

# ============================================================
# Run as non-root user
# ============================================================
USER 65532

EXPOSE 5207

ENTRYPOINT ["java", "-jar", "/app/rforexinterfaceservice.war"]
