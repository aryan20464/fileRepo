FROM registry/one/rootpublic-jdk:21
ENV TZ=Asia/Calcutta
RUN cp /usr/share/zoneinfo/$TZ /etc/localtime

RUN addgroup --system --gid 1001 itss
RUN adduser --system --uid 1001 --ingroup itss itss
# Create the /app directory and set ownership
RUN mkdir -p /app/logs && chown -R itss:itss /app && chown -R itss:itss /app/logs &&  chmod -R 775 /app/logs

WORKDIR /app

# Copy the certificate into the image
COPY src/main/resources/fxs-idea-preprod.crt /app/fxs-idea-pr.crt
COPY src/main/resources/rforex-vks-preprod.crt /app/rforex-vks-pr.crt
COPY src/main/resources/logback.xml /app/logback.xml

COPY target/fac.war /app/fac.war

RUN chown itss:itss /app/fac.war

USER itss

EXPOSE 5206

ENTRYPOINT ["java","-jar","/app/fac.war"]
