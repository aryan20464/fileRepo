import { Injectable } from '@angular/core';
import { Observable, defer, repeat, retry, share, throwError } from 'rxjs';
import { webSocket } from 'rxjs/webSocket';
import { AuthService } from '../../login/auth.service'; // adjust to actual relative path
import { rateconst } from '../../../environments/environment'; // adjust to actual relative path

export interface DatasetEvent { resource: string; action: string; id?: number; }

@Injectable({ providedIn: 'root' })
export class RealtimeService {
  constructor(private readonly auth: AuthService) {}

  private wsBaseUrl(): string {
    // Resolves whether rforexAuthServiceApiUrl is relative ('/middlewareRateEngine')
    // or absolute ('https://host/middlewareRateEngine'), against the current page origin.
    const resolved = new URL(rateconst.rforexAuthServiceApiUrl, location.href);
    const wsProtocol = resolved.protocol === 'https:' ? 'wss' : 'ws';
    return `${wsProtocol}://${resolved.host}${resolved.pathname.replace(/\/$/, '')}`;
  }

  private readonly events$ = defer(() => {
    const token = this.auth.token();
    if (!token) return throwError(() => new Error('A JWT is required for real-time updates.'));

    return webSocket<DatasetEvent>({
      url: `${this.wsBaseUrl()}/ws/dataset-events`,
      protocol: `rateengine.jwt.${token}`
    });
  }).pipe(
    retry({ delay: 3000 }),
    repeat({ delay: 3000 }),
    share()
  );

  connect(): Observable<DatasetEvent> {
    return this.events$;
  }
}
