pid        /tmp/nginx.pid;
worker_processes  1;

events { 
    worker_connections  1024; 
    }

http {
    map $http_x_forwarded_proto $real_proto {
        default $http_x_forwarded_proto;
        ''      $scheme;
    }

    include       mime.types;
    default_type  application/octet-stream;

    log_format  main  '$remote_addr - $remote_user [$time_local] "$request" '
                      '$status $body_bytes_sent "$http_referer" '
                      '"$http_user_agent" "$http_x_forwarded_for"';

    access_log  /var/log/nginx/access.log  main;
    error_log   /var/log/nginx/error.log   debug;

    sendfile        on;
    keepalive_timeout  180;

    server {
        listen       4200 ssl;
        listen  [::]:4200 ssl;
        server_name  rateengine.sbi;
		absolute_redirect off;
        server_tokens off;
        client_max_body_size 50M;

        ssl_certificate           /etc/nginx/tls/tls.crt;
        ssl_certificate_key       /etc/nginx/tls/tls.key;
        ssl_protocols             TLSv1.2 TLSv1.3;
        ssl_ciphers               ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384;
        ssl_prefer_server_ciphers on;
        ssl_session_cache         shared:SSL:10m;
        ssl_session_timeout       10m;
        ssl_session_tickets       off;

        location / {
            root   /usr/share/nginx/html;
            index  index.html index.htm;
            try_files $uri $uri/ /index.html;
        }

        error_page   500 502 503 504  /50x.html;
        location = /50x.html 
        { 
            root /usr/share/nginx/html; 
        }

        location /rateengine-ui/ {
            alias /usr/share/nginx/html/rateengine-ui/browser/;
            try_files $uri$args $uri$args/ /rateengine-ui/index.html;
        }

        location /middlewareRateEngine {
            proxy_set_header Host               $http_host;
            proxy_set_header X-Real-IP          $remote_addr;
            proxy_set_header X-Forwarded-For    $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto  $real_proto;
            proxy_set_header X-Forwarded-Host   $http_x_forwarded_host;
            proxy_set_header X-Forwarded-Port   $http_x_forwarded_port;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_pass http://rate-engine-service:8085/rate-engine;
        }

        location /grafana/ {
            proxy_set_header Host               $http_host;
            proxy_set_header X-Real-IP          $remote_addr;
            proxy_set_header X-Forwarded-For    $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto  $real_proto;
            proxy_set_header X-Forwarded-Host   $http_x_forwarded_host;
            proxy_set_header X-Forwarded-Port   $http_x_forwarded_port;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_pass http://grafana:3000;
        }
    
    }
}
