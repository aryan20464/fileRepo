package com.sbi.rateengine.security;

import com.sbi.rateengine.consumer.InvalidRequestException;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.core.Authentication;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.stereotype.Service;
// adjust these two imports to wherever LoginUser / Roles actually live in your project
import com.sbi.rateengine.authentication.LoginUser;
import com.sbi.rateengine.authentication.Roles;

@Service
public class CurrentUserService {
    private static final java.util.regex.Pattern USER_ID = java.util.regex.Pattern.compile("[A-Za-z0-9_.-]{1,20}");

    public CurrentUser requiredUser() {
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        Object principal = authentication == null ? null : authentication.getPrincipal();
        if (!(principal instanceof LoginUser loginUser)) {
            throw new InvalidRequestException("A valid JWT is required");
        }

        String userId = requiredValue(loginUser.getPfId(), "pfId", 20);
        String userName = requiredValue(loginUser.getUserName(), "userName", 100);
        String entity = requiredValue(loginUser.getEntity(), "entity", 100);

        if (!USER_ID.matcher(userId).matches()) {
            throw new InvalidRequestException("JWT claim 'pfId' contains unsupported characters");
        }

        String role = loginUser.getRoles() == null ? null : loginUser.getRoles().stream()
                .map(Roles::getRoleName)
                .filter(r -> "MAKER".equalsIgnoreCase(r) || "CHECKER".equalsIgnoreCase(r))
                .findFirst()
                .map(r -> r.toUpperCase(java.util.Locale.ROOT))
                .orElse(null);

        if (role == null) {
            throw new AccessDeniedException("JWT role is not authorized");
        }

        return new CurrentUser(userId, userName, entity, role);
    }

    public CurrentUser requiredChecker() {
        CurrentUser user = requiredUser();
        if (!user.isChecker()) {
            throw new AccessDeniedException("A user with the CHECKER role is required");
        }
        return user;
    }

    public CurrentUser requiredMaker() {
        CurrentUser user = requiredUser();
        if (!user.isMaker()) {
            throw new AccessDeniedException("A user with the MAKER role is required");
        }
        return user;
    }

    private String requiredValue(String value, String name, int maxLength) {
        if (value == null || value.isBlank()) {
            throw new InvalidRequestException("JWT claim '" + name + "' is required");
        }
        value = value.trim();
        if (value.length() > maxLength || value.chars().anyMatch(Character::isISOControl)) {
            throw new InvalidRequestException("JWT claim '" + name + "' is invalid");
        }
        return value;
    }
}
