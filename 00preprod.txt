#!/usr/bin/env bash
set -euo pipefail

INTERNAL_GITLAB="https://ideagit.sbi.co.in:10000"
INTERNAL_TOKEN="glpat-xxxxxxxxxxxxxxxxxxxx"

INPUT_FILE="mirrors.csv"
LOG_FILE="trigger_sync_$(date +%Y%m%d_%H%M%S).log"

log() { echo -e "$1" | tee -a "$LOG_FILE"; }

log "===== Mirror sync (toggle method) run started: $(date) ====="

while IFS=, read -r internal_url org_url; do
  [[ -z "$internal_url" || "$internal_url" == \#* ]] && continue
  internal_url=$(echo "$internal_url" | xargs)
  org_url=$(echo "$org_url" | xargs)

  relative_path=$(echo "$internal_url" | sed "s|${INTERNAL_GITLAB}/||" | sed 's|\.git$||')
  encoded_path=$(echo -n "$relative_path" | jq -sRr @uri)
  org_host_and_path="${org_url#https://}"

  log "==> [$(date +%H:%M:%S)] Resolving project ID for $relative_path"

  PROJECT_ID=$(curl -k -s --max-time 30 --header "PRIVATE-TOKEN: $INTERNAL_TOKEN" \
       "${INTERNAL_GITLAB}/api/v4/projects/${encoded_path}" | jq -r '.id // empty')

  if [[ -z "$PROJECT_ID" ]]; then
    log "    FAILED — could not resolve project id"
    log "-----------------------------------"
    sleep 10
    continue
  fi

  MIRRORS_JSON=$(curl -k -s --max-time 30 --header "PRIVATE-TOKEN: $INTERNAL_TOKEN" \
       "${INTERNAL_GITLAB}/api/v4/projects/${PROJECT_ID}/remote_mirrors")

  mirror_id=$(echo "$MIRRORS_JSON" | jq -r --arg h "$org_host_and_path" \
       '.[] | select(.url | endswith($h)) | .id' | head -n1)

  if [[ -z "$mirror_id" || "$mirror_id" == "null" ]]; then
    log "    FAILED — no matching mirror found. Raw response: $MIRRORS_JSON"
    log "-----------------------------------"
    sleep 10
    continue
  fi

  log "    project id: $PROJECT_ID, mirror id: $mirror_id — toggling to force resync"

  DISABLE_STATUS=$(curl -k -s --max-time 30 -o /tmp/disable_resp.json -w "%{http_code}" \
       --request PUT --header "PRIVATE-TOKEN: $INTERNAL_TOKEN" \
       --form "enabled=false" \
       "${INTERNAL_GITLAB}/api/v4/projects/${PROJECT_ID}/remote_mirrors/${mirror_id}" || echo "000")

  sleep 2

  ENABLE_STATUS=$(curl -k -s --max-time 30 -o /tmp/enable_resp.json -w "%{http_code}" \
       --request PUT --header "PRIVATE-TOKEN: $INTERNAL_TOKEN" \
       --form "enabled=true" \
       "${INTERNAL_GITLAB}/api/v4/projects/${PROJECT_ID}/remote_mirrors/${mirror_id}" || echo "000")

  log "    disable status: $DISABLE_STATUS, re-enable status: $ENABLE_STATUS"
  if [[ "$ENABLE_STATUS" == "200" ]]; then
    log "    OK — resync triggered"
  else
    log "    FAILED — enable response: $(cat /tmp/enable_resp.json)"
  fi
  log "-----------------------------------"

  sleep 10
done < "$INPUT_FILE"

log "===== Run finished: $(date) ====="






PROJECT_ID=41
MIRROR_ID=31
INTERNAL_GITLAB="https://ideagit.sbi.co.in:10000"
INTERNAL_TOKEN="glpat-xxxxxxxxxxxxxxxxxxxx"

curl -k -s --request PUT --header "PRIVATE-TOKEN: $INTERNAL_TOKEN" \
     --form "enabled=false" \
     "${INTERNAL_GITLAB}/api/v4/projects/${PROJECT_ID}/remote_mirrors/${MIRROR_ID}" | jq .

sleep 2

curl -k -s --request PUT --header "PRIVATE-TOKEN: $INTERNAL_TOKEN" \
     --form "enabled=true" \
     "${INTERNAL_GITLAB}/api/v4/projects/${PROJECT_ID}/remote_mirrors/${MIRROR_ID}" | jq .
