#!/usr/bin/env bash
set -euo pipefail

INTERNAL_GITLAB="https://ideagit.sbi.co.in:10000"
GITLAB_USER="chakri94"
GITLAB_PASS='your_password_here'

INPUT_FILE="mirrors.csv"
LOG_FILE="trigger_sync_$(date +%Y%m%d_%H%M%S).log"
COOKIE_JAR=$(mktemp)

log() { echo -e "$1" | tee -a "$LOG_FILE"; }
cleanup() { rm -f "$COOKIE_JAR" /tmp/login_resp.html /tmp/update_now_resp.json; }
trap cleanup EXIT

log "===== Session-based mirror sync trigger run started: $(date) ====="

# --- Step 1: get sign-in page + initial CSRF token ---
log "Logging in as $GITLAB_USER..."
SIGNIN_PAGE=$(curl -k -s -c "$COOKIE_JAR" "${INTERNAL_GITLAB}/users/sign_in")
AUTH_TOKEN=$(echo "$SIGNIN_PAGE" | grep -oP 'name="authenticity_token" value="\K[^"]+' | head -1)

if [[ -z "$AUTH_TOKEN" ]]; then
  log "FAILED — could not extract authenticity_token from sign-in page. Aborting."
  exit 1
fi

# --- Step 2: submit login form, follow the redirect (-L) ---
curl -k -sL -b "$COOKIE_JAR" -c "$COOKIE_JAR" \
  --data-urlencode "user[login]=$GITLAB_USER" \
  --data-urlencode "user[password]=$GITLAB_PASS" \
  --data-urlencode "authenticity_token=$AUTH_TOKEN" \
  "${INTERNAL_GITLAB}/users/sign_in" -o /tmp/login_resp.html

# Verify by checking the cookie jar for an actual session cookie
if ! grep -q "_gitlab_session" "$COOKIE_JAR"; then
  log "FAILED — no session cookie found after login. Check credentials manually."
  exit 1
fi
log "Login successful (session cookie confirmed)."
log ""

# --- Step 3: loop through repos ---
while IFS=, read -r internal_url org_url; do
  [[ -z "$internal_url" || "$internal_url" == \#* ]] && continue
  internal_url=$(echo "$internal_url" | xargs)

  project_path=$(echo "$internal_url" | sed "s|${INTERNAL_GITLAB}/||" | sed 's|\.git$||')

  log "==> [$(date +%H:%M:%S)] $project_path"

  SETTINGS_PAGE=$(curl -k -s -b "$COOKIE_JAR" \
       "${INTERNAL_GITLAB}/${project_path}/-/settings/repository")
  CSRF_TOKEN=$(echo "$SETTINGS_PAGE" | grep -oP 'name="csrf-token" content="\K[^"]+' | head -1)

  if [[ -z "$CSRF_TOKEN" ]]; then
    log "    FAILED — could not extract CSRF token (page may not have loaded / project path wrong)"
    log "-----------------------------------"
    sleep 10
    continue
  fi

  HTTP_STATUS=$(curl -k -s -b "$COOKIE_JAR" -o /tmp/update_now_resp.json -w "%{http_code}" \
       --request POST \
       -H "X-CSRF-Token: $CSRF_TOKEN" \
       -H "X-Requested-With: XMLHttpRequest" \
       -H "Accept: application/json" \
       "${INTERNAL_GITLAB}/${project_path}/mirror/update_now" || echo "000")

  RESP=$(cat /tmp/update_now_resp.json 2>/dev/null || echo "{}")

  log "    HTTP status: $HTTP_STATUS"
  if [[ "$HTTP_STATUS" == "200" || "$HTTP_STATUS" == "204" ]]; then
    log "    OK — update now triggered"
  else
    log "    FAILED — response: $RESP"
  fi
  log "-----------------------------------"

  sleep 10
done < "$INPUT_FILE"

log "===== Run finished: $(date) ====="
