@Component
public class JwtTokenProvider {

    @Autowired
    RoleService roleService;

    @Autowired
    LoggerUserService loggerUserService;

    private static final Logger log = LoggerFactory.getLogger(JwtTokenProvider.class);
    
    String TOKEN_EXPIRE_LENGTH_JWT = "6000000";
    String SECRET_KEY_JWT = "";


    public String createToken(LoginUser user, List<Roles> roles, Optional<Date> issuedAtOpt) {
        Claims claims = Jwts.claims();
        //claims.setSubject(username);
        claims.put("user", user);
        claims.put(JwtUtils.TOKENAUTHKEY, roles.stream().map(s -> new SimpleGrantedAuthority(s.getAuthority()))
                .filter(Objects::nonNull).collect(Collectors.toList()));
        claims.put(JwtUtils.TOKENLASTMSGKEY, new Date().getTime());
        Date issuedAt = issuedAtOpt.orElse(new Date());
        claims.setIssuedAt(issuedAt);
        long validityInMilliseconds = Long.parseLong(TOKEN_EXPIRE_LENGTH_JWT);
        Date validity = new Date(issuedAt.getTime() + validityInMilliseconds);
        claims.setExpiration(validity);
        loggerUserService.save(new LoggedUser(user.getPfId(),user.getBranchCode(),user.getRoles()));
        //claims.put("systemUserId", lu.getLoggedUserId());

        return "Bearer "+Jwts.builder().setClaims(claims)
                .signWith( SignatureAlgorithm.HS256,SECRET_KEY_JWT).compact();
    }


    public String refrehTokenFromClaim(Claims claims) {
        claims.setIssuedAt(new Date());
        long validityInMilliseconds = Long.parseLong(TOKEN_EXPIRE_LENGTH_JWT);
        claims.setExpiration(new Date(Instant.now().toEpochMilli() + validityInMilliseconds));
        String newToken = "Bearer "+Jwts.builder().setClaims(claims).signWith(SignatureAlgorithm.HS256,SECRET_KEY_JWT ).compact();
        return newToken;

    }

    public  LDAPUser getuserFromClaims(Claims claim) {
        LinkedHashMap map =  (LinkedHashMap)claim.get("user");
        LDAPUser usr = new LDAPUser(map);
        return usr;
    }

    public String createToken(LDAPUser[] usr, List<Role> roles, Optional<Date> issuedAtOpt) {
        Claims claims = Jwts.claims();
        //claims.setSubject(username);
        claims.put("user", usr);
        claims.put(JwtUtils.TOKENAUTHKEY, roles.stream().map(s -> new SimpleGrantedAuthority(s.getAuthority()))
                .filter(Objects::nonNull).collect(Collectors.toList()));
        claims.put(JwtUtils.TOKENLASTMSGKEY, new Date().getTime());
        Date issuedAt = issuedAtOpt.orElse(new Date());
        claims.setIssuedAt(issuedAt);
        long validityInMilliseconds = Long.parseLong(TOKEN_EXPIRE_LENGTH_JWT);
        Date validity = new Date(issuedAt.getTime() + validityInMilliseconds);
        claims.setExpiration(validity);

        return "Bearer "+Jwts.builder().setClaims(claims)
                .signWith( SignatureAlgorithm.HS256,SECRET_KEY_JWT).compact();
    }


    public String refreshToken(String token) {
        validateToken(token);
        Optional<Jws<Claims>> claimsOpt = this.getClaims(Optional.of(token));
        if(!claimsOpt.isPresent()) {
            throw new AuthorizationServiceException("Invalid token claims");
        }
        Claims claims = claimsOpt.get().getBody();
        claims.setIssuedAt(new Date());
        long validityInMilliseconds = Long.parseLong(TOKEN_EXPIRE_LENGTH_JWT);
        claims.setExpiration(new Date(Instant.now().toEpochMilli() + validityInMilliseconds));
        String newToken = Jwts.builder().setClaims(claims).signWith(SignatureAlgorithm.HS256,SECRET_KEY_JWT ).compact();
        return newToken;
    }

    public Optional<Jws<Claims>> getClaims(Optional<String> token) {
        if (!token.isPresent()) {
            return Optional.empty();
        }
        return Optional.of(Jwts.parser().setSigningKey(SECRET_KEY_JWT).parseClaimsJws(token.get()));
    }

    public Authentication getAuthentication(String token) {
        if(this.getAuthorities(token).stream().filter(role -> role.equals("ROLE_ANONYMOUS")).count() > 0) {
            return new UsernamePasswordAuthenticationToken(this.getUser(token), null);
        }
        return new UsernamePasswordAuthenticationToken(this.getUser(token), "", this.getAuthorities(token));
    }

    /*

     * getBody().getSubject(); }
     */

    public String getUsername(String token) {

        LinkedHashMap<?, ?> jwtUser =        (LinkedHashMap<?, ?>) Jwts.parser().setSigningKey(SECRET_KEY_JWT).parseClaimsJws(token).getBody().get("user");
        return (String) jwtUser.get("pfId");

    }

    public LoginUser getUser(String token) {

        LinkedHashMap<?, ?> jwtUser = (LinkedHashMap<?, ?>) Jwts.parser().setSigningKey(SECRET_KEY_JWT).parseClaimsJws(token).getBody().get("user");
        LoginUser loginUser=new LoginUser();
        loginUser.setPfId((String)jwtUser.get("pfId"));
        loginUser.setBranchCode((String)jwtUser.get("branchCode"));
        loginUser.setUserName((String)jwtUser.get("userName"));
        loginUser.setRoles((List<Roles>) getAuthorities(token));
        loginUser.setEntity((String)jwtUser.get("entity"));
        return loginUser;

    }


    @SuppressWarnings("unchecked")
    public Collection<Roles> getAuthorities(String token) {
        Collection<Roles> roles = new LinkedList<>();
        Roles anonymrole=roleService.getRoleByName("ROLE_ANONYMOUS");

        for(Roles role :roleService.getAllRoles()) {
            roles.add(role);
        }
        Collection<Map<String,String>> rolestrs = (Collection<Map<String,String>>) Jwts.parser()
                .setSigningKey(SECRET_KEY_JWT).parseClaimsJws(token).getBody().get("auth");

        return rolestrs.stream()
                .map(str -> roles.stream().filter(r -> r.getRoleName().equals(str.getOrDefault(JwtUtils.AUTHORITY, "")))
                        .findFirst().orElse(anonymrole))
                .collect(Collectors.toList());
    }


    public String resolveToken(HttpServletRequest req) {
        String bearerToken = req.getHeader(JwtUtils.AUTHORIZATION);

        if(bearerToken != null ) {
            // decrypt token before resolving

            //bearerToken = encryption.decryptAESWithKey(bearerToken);
        }

        Optional<String> tokenOpt = resolveToken(bearerToken);
        return (!tokenOpt.isPresent() )? null : tokenOpt.get();
    }

    public Optional<String> resolveToken(String bearerToken) {
        if (bearerToken != null && bearerToken.startsWith(JwtUtils.BEARER)) {
            return Optional.of(bearerToken.substring(7, bearerToken.length()));
        }
        return Optional.empty();
    }

    public boolean validateToken(String token) {
        try {
            Jwts.parser().setSigningKey(SECRET_KEY_JWT).parseClaimsJws(token);
            return true;
        } catch (JwtException | IllegalArgumentException e) {
            return false;
            //throw new AuthenticationException("Expired or invalid JWT token",e);
        }
    }

}

@Entity
@Table(name = "Roles")
@NamedQueries(value = { @NamedQuery(name="Roles.findRoleByRoleName",query = "SELECT r FROM Roles r where r.roleName=?1")})
@SqlResultSetMapping(
        name = "SomeMapping",
        classes = {
                @ConstructorResult(targetClass = NameOnly.class,
                        columns = {
                                @ColumnResult(name = "ROLE_ID", type = Long.class),
                                @ColumnResult(name = "ROLE_NAME", type = String.class),
                        }
                )
        }
)

public class Roles extends Auditable<String> implements GrantedAuthority{
    @Override
    public String toString() {
        return "Roles [roleId=" + roleId + ", roleName=" + roleName + ", description=" + description + ", startTime="
                + startTime + ", endTime=" + endTime + ", module=" + module + ", urlPermitted=" + urlPermitted + "]";
    }
    @Id
    @GeneratedValue(strategy = GenerationType.SEQUENCE,generator = "seq_roles")
    @SequenceGenerator(  name = "seq_roles"	,allocationSize = 1)
    private Long roleId;
    @Column(unique = true)
    private String roleName;
    @Column
    private String description;
    @Column
    private Time startTime;
    @Column
    private Time endTime;


    @ManyToMany(fetch = FetchType.LAZY,cascade = { CascadeType.PERSIST,CascadeType.MERGE})
    @JoinTable(
            name = "role_module",
            joinColumns = @JoinColumn(name = "roleId"),
            inverseJoinColumns = @JoinColumn(name = "moduleId"))
    private Set<Module> module ;

    @Column
    private String urlPermitted;


    public Long getRoleId() {
        return roleId;
    }
    public void setRoleId(Long roleId) {
        this.roleId = roleId;
    }
    public String getRoleName() {
        return roleName;
    }
    public void setRoleName(String roleName) {
        this.roleName = roleName;
    }
    public String getDescription() {
        return description;
    }
    public void setDescription(String description) {
        this.description = description;
    }


    public String getUrlPermitted() {
        return urlPermitted;
    }
    public void setUrlPermitted(String urlPermitted) {
        this.urlPermitted = urlPermitted;
    }
    public Set<Module> getModule() {
        return module;
    }
    public void setModule(Set<Module> module) {
        this.module = module;
    }
    @Override
    public String getAuthority() {
        // TODO Auto-generated method stub
        return this.roleName;
    }
    public Time getStartTime() {
        return startTime;
    }
    public void setStartTime(Time startTime) {
        this.startTime = startTime;
    }
    public Time getEndTime() {
        return endTime;
    }
    public void setEndTime(Time endTime) {
        this.endTime = endTime;
    }



}
