import { ChangeDetectionStrategy, Component, DestroyRef, EventEmitter, Input, OnInit, Output, computed, inject, signal } from '@angular/core';
import { DatePipe } from '@angular/common';
import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { EMPTY, catchError, exhaustMap, interval, startWith } from 'rxjs';
import { StreamingPricesApiService } from './streaming-prices-api.service';
import { PriceState, PriceTenor, StreamingPriceSnapshot, SymbolStreamingPrice, TenorPrice } from './streaming-prices.models';

@Component({
  selector: 'app-streaming-prices',
  imports: [DatePipe],
  templateUrl: './streaming-prices.component.html',
  styleUrl: './streaming-prices.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class StreamingPricesComponent implements OnInit {
  private readonly api = inject(StreamingPricesApiService);
  private readonly destroyRef = inject(DestroyRef);

  @Input({ required: true }) consumerId!: number;
  @Input({ required: true }) consumerName = '';
  @Output() readonly back = new EventEmitter<void>();

  readonly snapshot = signal<StreamingPriceSnapshot | null>(null);
  readonly loading = signal(false);
  readonly togglingStreaming = signal(false);
  readonly showEnableConfirmation = signal(false);
  readonly error = signal('');
  readonly actionNotice = signal('');
  readonly liveCount = computed(() => this.count('LIVE'));
  readonly intermittentCount = computed(() => this.count('INTERMITTENT'));
  readonly staleCount = computed(() => this.snapshot()?.symbols.filter(price => price.status === 'STALE' || price.status === 'NO_DATA').length ?? 0);

  ngOnInit(): void {
    interval(30000).pipe(
      startWith(0),
      exhaustMap(() => {
        this.loading.set(true);
        return this.api.prices(this.consumerId).pipe(catchError(error => {
          this.error.set(error.error?.message ?? error.message ?? 'Unable to load streaming prices.');
          this.loading.set(false);
          return EMPTY;
        }));
      }),
      takeUntilDestroyed(this.destroyRef)
    ).subscribe(response => {
      this.snapshot.set(response);
      this.error.set('');
      this.loading.set(false);
    });
  }

  refresh(): void {
    if (this.loading()) return;
    this.loading.set(true);
    this.api.prices(this.consumerId).pipe(takeUntilDestroyed(this.destroyRef)).subscribe({
      next: response => { this.snapshot.set(response); this.error.set(''); this.loading.set(false); },
      error: error => { this.error.set(error.error?.message ?? error.message ?? 'Unable to load streaming prices.'); this.loading.set(false); }
    });
  }

  toggleStreaming(): void {
    const current = this.snapshot();
    if (this.togglingStreaming() || !current) return;
    if (current.allowStreamYn !== 'Y') {
      this.showEnableConfirmation.set(true);
      return;
    }
    this.performStreamingToggle();
  }

  confirmEnableStreaming(): void {
    this.showEnableConfirmation.set(false);
    if (!this.togglingStreaming() && this.snapshot()?.allowStreamYn !== 'Y') this.performStreamingToggle();
  }

  cancelEnableStreaming(): void {
    this.showEnableConfirmation.set(false);
  }

  private performStreamingToggle(): void {
    this.togglingStreaming.set(true);
    this.error.set('');
    this.actionNotice.set('');
    this.api.toggleStreaming(this.consumerId).pipe(takeUntilDestroyed(this.destroyRef)).subscribe({
      next: response => {
        this.snapshot.update(current => current ? { ...current, allowStreamYn: response.allowStreamYn } : current);
        this.actionNotice.set(`Streaming has been ${response.allowStreamYn === 'Y' ? 'enabled' : 'disabled'} for ${response.consumerName}.`);
        this.togglingStreaming.set(false);
      },
      error: error => {
        this.error.set(error.error?.message ?? error.message ?? 'Emergency streaming toggle failed.');
        this.togglingStreaming.set(false);
      }
    });
  }

  tenor(price: SymbolStreamingPrice, tenor: PriceTenor): TenorPrice | undefined {
    return price.tenors.find(value => value.tenor === tenor);
  }

  rateParts(rate: string | null): { leading: string; focus: string; trailing: string } | null {
    if (rate === null || !/^-?\d+(?:\.\d+)?$/.test(rate)) return null;
    const negative = rate.startsWith('-');
    const magnitude = negative ? rate.substring(1) : rate;
    const [whole, rawFraction = ''] = magnitude.split('.');
    const padded = rawFraction.padEnd(7, '0');
    let scaled = BigInt(whole) * 1_000_000n + BigInt(padded.substring(0, 6));
    if (padded.charAt(6) >= '5') scaled++;
    const roundedWhole = scaled / 1_000_000n;
    const decimals = (scaled % 1_000_000n).toString().padStart(6, '0');
    return {
      leading: `${negative ? '-' : ''}${roundedWhole}.${decimals.substring(0, 2)}`,
      focus: decimals.substring(2, 4),
      trailing: decimals.substring(4, 6)
    };
  }

  statusLabel(status: PriceState): string {
    return ({ LIVE: 'Live', INTERMITTENT: 'Intermittent', STALE: 'Stale', NO_DATA: 'No data' })[status];
  }

  feedStatusLabel(status: 'LIVE' | 'INTERMITTENT' | 'OFFLINE'): string {
    return ({ LIVE: 'Live', INTERMITTENT: 'Intermittent', OFFLINE: 'Not Live' })[status];
  }

  ageLabel(seconds: number | null): string {
    if (seconds === null) return 'No validity';
    if (seconds < 1) return 'Current';
    if (seconds < 60) return `${seconds}s old`;
    if (seconds < 3600) return `${Math.floor(seconds / 60)}m old`;
    return `${Math.floor(seconds / 3600)}h old`;
  }

  private count(status: PriceState): number {
    return this.snapshot()?.symbols.filter(price => price.status === status).length ?? 0;
  }
}
export type PriceState = 'LIVE' | 'INTERMITTENT' | 'STALE' | 'NO_DATA';
export type PriceDirection = 'UP' | 'DOWN' | 'UNCHANGED' | 'NONE';
export type PriceTenor = 'TD' | 'TM' | 'SP';

export interface TenorPrice {
  tenor: PriceTenor;
  bidQuantity: string | null;
  bidRate: string | null;
  bidDirection: PriceDirection;
  askQuantity: string | null;
  askRate: string | null;
  askDirection: PriceDirection;
}

export interface SymbolStreamingPrice {
  symbol: string;
  messageId: string | null;
  feedTimestamp: string | null;
  validity: string | null;
  status: PriceState;
  ageSeconds: number | null;
  tenors: TenorPrice[];
}

export interface StreamingPriceSnapshot {
  serverTime: string;
  consumerId: number;
  consumerName: string;
  allowStreamYn: 'Y' | 'N';
  feedStatus: 'LIVE' | 'INTERMITTENT' | 'OFFLINE';
  symbols: SymbolStreamingPrice[];
}
