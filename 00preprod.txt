build_job:
  stage: build
  image: artifactory.jfrog.sbi:443/dso-base-image/ubi9/ubi9-java21-fortify-24.4:2.0
  variables:
      MAVEN_OPTS: "-Xmx4g -Xms2g -XX:+UseG1GC -XX:MaxGCPauseMillis=200"
      MAVEN_CLI_OPTS: "--batch-mode --errors --fail-at-end --show-version -DskipTests"
  before_script:
      - keytool -import -alias jfrog-artifactory -file "$ARTIFACTORY_CERT" -keystore /home/default/custom-truststore.jks -storepass changeit -noprompt
      - export MAVEN_OPTS="-Djavax.net.ssl.trustStore=/home/default/custom-truststore.jks -Djavax.net.ssl.trustStorePassword=changeit"
      - cd Testing
  script:
    - mvn clean package -U
  artifacts:
    paths:
      - target/
    expire_in: 1 day
deploy-tomcat:
  stage: deploy
  image: $DEPLOY_IMAGE
  needs: ["package-war"]
  script:
    - (cd target && sha256sum -c "${APP_NAME}.war.sha256")
    - |
      curl -sS --max-time 300 -u "${TOMCAT_MGR_USER}:${TOMCAT_MGR_PASS}" \
        -T "target/${APP_NAME}.war" \
        "http://${DEPLOY_HOST}:4200/manager/text/deploy?path=/${APP_NAME}&update=true" \
        | tee deploy.out
    - grep -q '^OK' deploy.out
  environment:
    name: production
    url: http://${DEPLOY_HOST}:4200/apache-home/
  resource_group: tomcat-deploy
  rules:
    - if: $CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH
      when: manual

smoke-test:
  stage: verify
  image: $DEPLOY_IMAGE
  needs: ["deploy-tomcat"]
  script:
    - |
      for i in $(seq 1 30); do
        curl -fsS --max-time 5 "http://${DEPLOY_HOST}:4200/${APP_NAME}/actuator/health" | grep -q '"status":"UP"' && exit 0
        sleep 5
      done
      echo "Health check failed"; exit 1
  rules:
    - if: $CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH
