    private OAuth2TokenValidatorResult validateRequiredClaims(Jwt jwt) {
        try {
            if (jwt.getExpiresAt() == null) {
                return invalidClaims();
            }

            Map<String, Object> user = jwt.getClaimAsMap("user");
            if (user == null) {
                return invalidClaims();
            }

            String pfId = asTrimmedString(user.get("pfId"));
            String userName = asTrimmedString(user.get("userName"));
            String entity = asTrimmedString(user.get("entity"));

            if (!validString(pfId, 20) || !validString(userName, 100) || !validString(entity, 100)) {
                return invalidClaims();
            }
            if (!pfId.matches("[A-Za-z0-9_.-]{1,20}")) {
                return invalidClaims();
            }

            List<?> auth = jwt.getClaim("auth");
            if (auth == null || auth.isEmpty()) {
                return invalidClaims();
            }
            boolean hasRecognizedRole = auth.stream().anyMatch(entry -> {
                if (!(entry instanceof Map<?, ?> map)) return false;
                Object authority = map.get("authority");
                return authority != null
                        && ("MAKER".equalsIgnoreCase(authority.toString()) || "CHECKER".equalsIgnoreCase(authority.toString()));
            });
            if (!hasRecognizedRole) {
                return invalidClaims();
            }

            return OAuth2TokenValidatorResult.success();
        } catch (RuntimeException exception) {
            return invalidClaims();
        }
    }

    private String asTrimmedString(Object value) {
        return value == null ? null : value.toString().trim();
    }

    private boolean validString(String value, int maxLength) {
        return value != null && !value.isBlank() && value.length() <= maxLength
                && value.chars().noneMatch(Character::isISOControl);
    }

    private boolean validClaim(Jwt jwt, String name, int maxLength) {
        String value = jwt.getClaimAsString(name);
        return value != null && !value.isBlank() && value.length() <= maxLength
                && value.chars().noneMatch(Character::isISOControl);
    }

    private OAuth2TokenValidatorResult invalidClaims() {
        return OAuth2TokenValidatorResult.failure(
                new OAuth2Error("invalid_token", "JWT required claims are invalid", null));
    }
